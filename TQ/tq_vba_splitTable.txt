Option Explicit

Dim Th1(500) As Double, Th2(500) As Double, qh1(500) As Double, Th(500) As Double, Qh(500) As Double
Dim Tl1(500) As Double, Tl2(500) As Double, ql1(500) As Double, Tl(500) As Double, Ql(500) As Double
Dim dTh(500) As Double, dTl(500) As Double, dThl(500) As Double
Dim forTGC(500) As Double, TGC(500) As Double, GCC(500) As Double, GCCmin As Double
Dim TGCThmin As Double, TGCTcmax As Double
Dim i As Integer, j As Long, NHdata As Integer, NLdata As Integer, GCdata As Integer
Dim MINdT As Double, minTh As Double, shiftQ As Double
Sub TQDiagram()
Dim Col_Qex As Integer, Col_Qen As Integer, Col_Data As Integer, Col_Cal As Integer
Const EPS_T As Double = 0.001
Const EPS_Q As Double = 0.000001
Const EPS_Q_OUT As Double = 0.000001

With ActiveSheet

    .Range("W4:AE100").ClearContents

    Col_Data = 3
    Col_Qex = 7
    Col_Qen = 15
    Col_Cal = 23

    .Range("F4:K43").Sort Key1:=.Range("I4"), Order1:=xlAscending, Header:=xlNo
    .Range("N4:S43").Sort Key1:=.Range("O4"), Order1:=xlAscending, Header:=xlNo

NHdata = .Cells(3, Col_Data).Value
NLdata = .Cells(4, Col_Data).Value
MINdT = .Cells(5, Col_Data).Value

For i = 1 To NHdata
    Th2(i) = Round(.Cells(i + 3, Col_Qex).Value, 3)
    Th1(i) = Round(.Cells(i + 3, Col_Qex + 2).Value, 3)
    If Th2(i) < Th1(i) Then
        MsgBox "The hot stream temperature input is invalid." & vbCrLf & "The outlet temperature must be less than or equal to the inlet temperature.", vbCritical
        Exit Sub
    End If
    qh1(i) = .Cells(i + 3, Col_Qex + 3).Value
    .Cells(2 * i + 2, Col_Cal + 4) = Th1(i)
    .Cells(2 * i + 3, Col_Cal + 4) = Th2(i)
Next i

For i = 1 To NLdata
    Tl1(i) = Round(.Cells(i + 3, Col_Qen).Value, 3)
    Tl2(i) = Round(.Cells(i + 3, Col_Qen + 2).Value, 3)
    If Tl2(i) < Tl1(i) Then
        MsgBox "The cold stream temperature input is invalid." & vbCrLf & "The outlet temperature must be greater than or equal to the inlet temperature.", vbCritical
        Exit Sub
    End If
    ql1(i) = .Cells(i + 3, Col_Qen + 3).Value
    .Cells(2 * i + 2, Col_Cal + 7) = Tl1(i)
    .Cells(2 * i + 3, Col_Cal + 7) = Tl2(i)
Next i

    .Range("AA4:AA100").Sort Key1:=.Range("AA4"), Order1:=xlAscending, Header:=xlNo
    .Range("AD4:AD100").Sort Key1:=.Range("AD4"), Order1:=xlAscending, Header:=xlNo

For i = 1 To NHdata * 2
    Th(i) = .Cells(i + 3, Col_Cal + 4).Value
    forTGC(i) = Th(i)
Next i

For i = 1 To NLdata * 2
    Tl(i) = .Cells(i + 3, Col_Cal + 7).Value
    forTGC(NHdata * 2 + i) = Tl(i) + MINdT
Next i

TGCThmin = .Cells(4, Col_Cal + 4).Value
TGCTcmax = .Cells(3 + NLdata * 2, Col_Cal + 7).Value

Qh(1) = 0
.Cells(4, Col_Cal + 5).Value = Qh(1)
For i = 2 To NHdata * 2
    Qh(i) = Qh(i - 1)
    If Th(i) <> Th(i - 1) Then
        For j = 1 To NHdata
            If (Th1(j) <> Th2(j)) And (Th(i) > Th1(j)) And (Th(i) <= Th2(j)) Then
                Qh(i) = Qh(i) + qh1(j) / (Th2(j) - Th1(j)) * (Th(i) - Th(i - 1))
            End If
        Next j
    ElseIf Th(i) = Th(i - 1) Then
        For j = 1 To NHdata
            If (Th1(j) = Th2(j)) And (Th1(j) = Th(i)) Then
                Qh(i) = Qh(i) + qh1(j)
            End If
        Next j
    End If
    .Cells(i + 3, Col_Cal + 5).Value = Qh(i)
Next i

Ql(1) = 0
.Cells(4, Col_Cal + 8).Value = Ql(1)
For i = 2 To NLdata * 2
    Ql(i) = Ql(i - 1)
    If Tl(i) <> Tl(i - 1) Then
    For j = 1 To NLdata
        If (Tl1(j) <> Tl2(j)) And (Tl(i) > Tl1(j)) And (Tl(i) <= Tl2(j)) Then
        Ql(i) = Ql(i) + ql1(j) / (Tl2(j) - Tl1(j)) * (Tl(i) - Tl(i - 1))
        End If
    Next j
    ElseIf Tl(i) = Tl(i - 1) Then
    For j = 1 To NLdata
        If (Tl1(j) = Tl2(j)) And (Tl1(j) = Tl(i)) Then
        Ql(i) = Ql(i) + ql1(j)
        End If
    Next j
    End If
    .Cells(i + 3, Col_Cal + 8).Value = Ql(i)
Next i

For i = 1 To NLdata * 2 + NHdata * 2
    .Cells(i + 3, Col_Cal).Value = forTGC(i)
Next i

.Range("W4:W100").Sort Key1:=.Range("W4"), Order1:=xlDescending, Header:=xlNo

j = 1
For i = 1 To NLdata * 2 + NHdata * 2
    forTGC(i) = .Cells(i + 3, Col_Cal).Value
    If j = 1 Then
        TGC(j) = forTGC(i)
        j = j + 1
    ElseIf forTGC(i) <> forTGC(i - 1) Then
        TGC(j) = forTGC(i)
        j = j + 1
    End If
Next i
GCdata = j - 1

.Range("W4:W100").ClearContents
Dim sen As Integer, sennetsu As Integer
sen = 0
sennetsu = 0
For j = 1 To GCdata + sen - 1
dTh(j) = 0
dTl(j) = 0
    For i = 1 To NHdata
        If Th2(i) <> Th1(i) And Th2(i) = TGC(j) Then
            dTh(j) = dTh(j) + (Th2(i) - TGC(j + 1)) / (Th2(i) - Th1(i)) * qh1(i)
        ElseIf Th2(i) = Th1(i) And Th2(i) = TGC(j) Then
            dTh(j) = dTh(j) + qh1(i)
            .Cells(j + sen + 3, Col_Cal).Value = TGC(j)
            sennetsu = 1
        ElseIf Th2(i) > TGC(j) And Th1(i) < TGC(j) Then
            dTh(j) = dTh(j) + (TGC(j) - TGC(j + 1)) / (Th2(i) - Th1(i)) * qh1(i)
        End If
    Next i
    For i = 1 To NLdata
        If Tl2(i) <> Tl1(i) And Tl2(i) + MINdT = TGC(j) Then
            dTl(j) = dTl(j) + (Tl2(i) + MINdT - TGC(j + 1)) / (Tl2(i) - Tl1(i)) * ql1(i)
        ElseIf Tl2(i) = Tl1(i) And Tl2(i) + MINdT = TGC(j) Then
            dTl(j) = dTl(j) + ql1(i)
            .Cells(j + sen + 3, Col_Cal).Value = TGC(j)
            sennetsu = 1
        ElseIf Tl2(i) + MINdT > TGC(j) And Tl1(i) + MINdT < TGC(j) Then
            dTl(j) = dTl(j) + (TGC(j) - TGC(j + 1)) / (Tl2(i) - Tl1(i)) * ql1(i)
        End If
    Next i
    If sennetsu = 0 Then
        dThl(j) = dTh(j) - dTl(j)
    Else
        dThl(j) = dTh(j) - dTl(j)
        .Cells(j + sen + 3, Col_Cal + 1).Value = dThl(j)
        sen = sen + 1
        dThl(j) = 0
        sennetsu = 0
    End If
    .Cells(j + sen + 3, Col_Cal).Value = TGC(j)
    .Cells(j + sen + 3, Col_Cal + 1).Value = dThl(j)
Next j

Dim Gccmin2 As Double, minth2 As Double, hasGccmin2 As Boolean
GCCmin = 0
minTh = 0
minth2 = 0
hasGccmin2 = False
GCC(0) = 0
For j = 1 To GCdata + sen - 2
    If TGC(j + 1) >= TGCThmin And TGC(j + 1) <= TGCTcmax + MINdT Then
        GCC(j) = GCC(j - 1) + .Cells(j + 3, Col_Cal + 1).Value
        If GCC(j) <= GCCmin Then
            GCCmin = GCC(j)
            minTh = .Cells(3 + j + 1, Col_Cal).Value
        End If
        If minTh = 0 And GCC(j) <= GCC(j - 1) Then
            Gccmin2 = GCC(j)
            minth2 = .Cells(3 + j + 1, Col_Cal).Value
            hasGccmin2 = True
        End If
    Else
        GCC(j) = GCC(j - 1) + .Cells(j + 3, Col_Cal + 1).Value
    End If
Next j

If minTh = 0 Then
    If hasGccmin2 Then
        minTh = minth2
        GCCmin = Gccmin2
    Else
        minTh = Tl(NLdata * 2) + MINdT
        GCCmin = dThl(1)
    End If
    GCC(0) = -GCC(1)
Else
    GCC(0) = 0
End If

For j = 0 To GCdata + sen - 2
    .Cells(j + 4, Col_Cal + 2).Value = GCC(j) + Abs(GCCmin)
Next j

shiftQ = Abs(GCCmin)


For i = 1 To NLdata * 2
    Ql(i) = Ql(i) + shiftQ
    .Cells(i + 3, Col_Cal + 8).Value = Ql(i)
Next i

Dim m As Integer, ii As Integer, NhN As Integer, NlN As Integer
Dim Nii As Integer, NiiP(500) As Double
Dim qStart As Double, qEnd As Double, qSeg As Double
Dim HotCompCount(500) As Integer, ColdCompCount(500) As Integer
Dim HotCompStreamNo(500, 100) As Integer, ColdCompStreamNo(500, 100) As Integer
Dim HotCompQdT(500, 100) As Double, ColdCompQdT(500, 100) As Double
Dim HotCompQdTsum(500) As Double, ColdCompQdTsum(500) As Double
Dim kHot As Integer, kCold As Integer, coldSeg As Integer
Dim nBounds As Integer, b As Integer, bb As Integer, hIdx As Integer, cIdx As Integer
Dim Bound(500) As Double, HotCum(100) As Double, ColdCum(100) As Double
Dim qPieceStart As Double, qPieceEnd As Double, qMid As Double, tmpQ As Double
Dim hotTStart As Double, hotTEnd As Double, coldTStart As Double, coldTEnd As Double
Dim hotPieceTStart As Double, hotPieceTEnd As Double, coldPieceTStart As Double, coldPieceTEnd As Double
Dim Col_Qex2 As Integer, Col_Qen2 As Integer, Col_Data2 As Integer

    Col_Data2 = 34
    Col_Qex2 = 38
    Col_Qen2 = 47

    .Range("AL4:AS200").ClearContents
    .Range("AU4:BB200").ClearContents

Th(0) = 0
Qh(0) = 0
j = 1
For i = 1 To NHdata * 2
    Th(j) = Round(.Cells(i + 3, Col_Cal + 4).Value, 3)
    Qh(j) = .Cells(i + 3, Col_Cal + 5).Value
    If Abs(Th(j) - Th(j - 1)) < EPS_T And Abs(Qh(j) - Qh(j - 1)) < EPS_Q Then
        j = j - 1
    End If
    j = j + 1
Next i
NhN = j - 1

Tl(0) = 0
Ql(0) = 0
j = 1
For i = 1 To NLdata * 2
    Tl(j) = Round(.Cells(i + 3, Col_Cal + 7).Value, 3)
    Ql(j) = .Cells(i + 3, Col_Cal + 8).Value
    If Abs(Tl(j) - Tl(j - 1)) < EPS_T And Abs(Ql(j) - Ql(j - 1)) < EPS_Q Then
        j = j - 1
    End If
    j = j + 1
Next i
NlN = j - 1

' Store the stream composition of each composite-curve segment once.
For i = 1 To NhN - 1
    HotCompCount(i) = 0
    HotCompQdTsum(i) = 0
    If Abs(Qh(i + 1) - Qh(i)) >= EPS_Q Then
        For j = 1 To NHdata
            If Abs(Th2(j) - Th1(j)) >= EPS_T Then
                If Th1(j) <= Th(i) + EPS_T And Th2(j) >= Th(i + 1) - EPS_T Then
                    HotCompCount(i) = HotCompCount(i) + 1
                    HotCompStreamNo(i, HotCompCount(i)) = j
                    HotCompQdT(i, HotCompCount(i)) = qh1(j) / (Th2(j) - Th1(j))
                    HotCompQdTsum(i) = HotCompQdTsum(i) + HotCompQdT(i, HotCompCount(i))
                End If
            ElseIf Abs(Th1(j) - Th(i)) < EPS_T And Abs(Th2(j) - Th(i + 1)) < EPS_T Then
                HotCompCount(i) = HotCompCount(i) + 1
                HotCompStreamNo(i, HotCompCount(i)) = j
                HotCompQdT(i, HotCompCount(i)) = qh1(j)
                HotCompQdTsum(i) = HotCompQdTsum(i) + HotCompQdT(i, HotCompCount(i))
            End If
        Next j
    End If
Next i

For i = 1 To NlN - 1
    ColdCompCount(i) = 0
    ColdCompQdTsum(i) = 0
    If Abs(Ql(i + 1) - Ql(i)) >= EPS_Q Then
        For j = 1 To NLdata
            If Abs(Tl2(j) - Tl1(j)) >= EPS_T Then
                If Tl1(j) <= Tl(i) + EPS_T And Tl2(j) >= Tl(i + 1) - EPS_T Then
                    ColdCompCount(i) = ColdCompCount(i) + 1
                    ColdCompStreamNo(i, ColdCompCount(i)) = j
                    ColdCompQdT(i, ColdCompCount(i)) = ql1(j) / (Tl2(j) - Tl1(j))
                    ColdCompQdTsum(i) = ColdCompQdTsum(i) + ColdCompQdT(i, ColdCompCount(i))
                End If
            ElseIf Abs(Tl1(j) - Tl(i)) < EPS_T And Abs(Tl2(j) - Tl(i + 1)) < EPS_T Then
                ColdCompCount(i) = ColdCompCount(i) + 1
                ColdCompStreamNo(i, ColdCompCount(i)) = j
                ColdCompQdT(i, ColdCompCount(i)) = ql1(j)
                ColdCompQdTsum(i) = ColdCompQdTsum(i) + ColdCompQdT(i, ColdCompCount(i))
            End If
        Next j
    End If
Next i

    kHot = 1
    kCold = 1
    For i = 1 To NhN - 1
        If Abs(Qh(i + 1) - Qh(i)) < EPS_Q Then GoTo ContinueHotSegment

        Nii = 0
        NiiP(0) = 0
        For ii = 1 To NlN
            If Abs(Ql(ii) - Ql(ii - 1)) >= EPS_Q And Qh(i) + EPS_Q < Ql(ii) And Ql(ii) < Qh(i + 1) - EPS_Q Then
                Nii = Nii + 1
                NiiP(Nii) = (Ql(ii) - Qh(i)) / (Qh(i + 1) - Qh(i))
            ElseIf Ql(ii) >= Qh(i + 1) - EPS_Q Then
                Exit For
            End If
        Next ii
        NiiP(Nii + 1) = 1

        If HotCompCount(i) > 0 And Abs(HotCompQdTsum(i)) >= EPS_Q Then
            For ii = 0 To Nii
                qStart = Qh(i) + (Qh(i + 1) - Qh(i)) * NiiP(ii)
                qEnd = Qh(i) + (Qh(i + 1) - Qh(i)) * NiiP(ii + 1)
                If Abs(qEnd - qStart) >= EPS_Q_OUT Then
                    coldSeg = 0
                    For j = 1 To NlN - 1
                        If Abs(Ql(j + 1) - Ql(j)) >= EPS_Q Then
                            If Ql(j) <= qStart + EPS_Q And qEnd <= Ql(j + 1) + EPS_Q Then
                                coldSeg = j
                                Exit For
                            End If
                        End If
                    Next j

                    If coldSeg > 0 And ColdCompCount(coldSeg) > 0 And Abs(ColdCompQdTsum(coldSeg)) >= EPS_Q Then
                        hotTStart = Round(Th(i) + (Th(i + 1) - Th(i)) * NiiP(ii), 3)
                        hotTEnd = Round(Th(i) + (Th(i + 1) - Th(i)) * NiiP(ii + 1), 3)
                        coldTStart = Round(Tl(coldSeg) + (Tl(coldSeg + 1) - Tl(coldSeg)) * (qStart - Ql(coldSeg)) / (Ql(coldSeg + 1) - Ql(coldSeg)), 3)
                        coldTEnd = Round(Tl(coldSeg) + (Tl(coldSeg + 1) - Tl(coldSeg)) * (qEnd - Ql(coldSeg)) / (Ql(coldSeg + 1) - Ql(coldSeg)), 3)

                        nBounds = 0
                        HotCum(0) = qStart
                        For m = 1 To HotCompCount(i)
                            HotCum(m) = HotCum(m - 1) + HotCompQdT(i, m) / HotCompQdTsum(i) * (qEnd - qStart)
                            nBounds = nBounds + 1
                            Bound(nBounds) = HotCum(m - 1)
                        Next m
                        HotCum(HotCompCount(i)) = qEnd
                        nBounds = nBounds + 1
                        Bound(nBounds) = qEnd

                        ColdCum(0) = qStart
                        For m = 1 To ColdCompCount(coldSeg)
                            ColdCum(m) = ColdCum(m - 1) + ColdCompQdT(coldSeg, m) / ColdCompQdTsum(coldSeg) * (qEnd - qStart)
                            nBounds = nBounds + 1
                            Bound(nBounds) = ColdCum(m - 1)
                        Next m
                        ColdCum(ColdCompCount(coldSeg)) = qEnd

                        For b = 1 To nBounds - 1
                            For bb = b + 1 To nBounds
                                If Bound(bb) < Bound(b) Then
                                    tmpQ = Bound(b)
                                    Bound(b) = Bound(bb)
                                    Bound(bb) = tmpQ
                                End If
                            Next bb
                        Next b

                        For b = 1 To nBounds - 1
                            qPieceStart = Bound(b)
                            qPieceEnd = Bound(b + 1)
                            If Abs(qPieceEnd - qPieceStart) >= EPS_Q_OUT Then
                                qMid = (qPieceStart + qPieceEnd) / 2
                                hIdx = 0
                                For m = 1 To HotCompCount(i)
                                    If HotCum(m - 1) <= qMid + EPS_Q And qMid <= HotCum(m) + EPS_Q Then
                                        hIdx = m
                                        Exit For
                                    End If
                                Next m
                                cIdx = 0
                                For m = 1 To ColdCompCount(coldSeg)
                                    If ColdCum(m - 1) <= qMid + EPS_Q And qMid <= ColdCum(m) + EPS_Q Then
                                        cIdx = m
                                        Exit For
                                    End If
                                Next m

                                If hIdx > 0 And cIdx > 0 Then
                                    hotPieceTStart = Round(hotTStart + (hotTEnd - hotTStart) * (qPieceStart - HotCum(hIdx - 1)) / (HotCum(hIdx) - HotCum(hIdx - 1)), 3)
                                    hotPieceTEnd = Round(hotTStart + (hotTEnd - hotTStart) * (qPieceEnd - HotCum(hIdx - 1)) / (HotCum(hIdx) - HotCum(hIdx - 1)), 3)
                                    coldPieceTStart = Round(coldTStart + (coldTEnd - coldTStart) * (qPieceStart - ColdCum(cIdx - 1)) / (ColdCum(cIdx) - ColdCum(cIdx - 1)), 3)
                                    coldPieceTEnd = Round(coldTStart + (coldTEnd - coldTStart) * (qPieceEnd - ColdCum(cIdx - 1)) / (ColdCum(cIdx) - ColdCum(cIdx - 1)), 3)
                                    qSeg = qPieceEnd - qPieceStart

                                    .Cells(3 + kHot, Col_Qex2) = HotCompStreamNo(i, hIdx)
                                    .Cells(3 + kHot, Col_Qex2 + 1) = qPieceStart
                                    .Cells(3 + kHot, Col_Qex2 + 2) = qPieceEnd
                                    .Cells(3 + kHot, Col_Qex2 + 3) = hotPieceTStart
                                    .Cells(3 + kHot, Col_Qex2 + 4) = hotPieceTEnd
                                    .Cells(3 + kHot, Col_Qex2 + 5) = qSeg
                                    kHot = kHot + 1

                                    .Cells(3 + kCold, Col_Qen2) = ColdCompStreamNo(coldSeg, cIdx)
                                    .Cells(3 + kCold, Col_Qen2 + 1) = qPieceStart
                                    .Cells(3 + kCold, Col_Qen2 + 2) = qPieceEnd
                                    .Cells(3 + kCold, Col_Qen2 + 3) = coldPieceTStart
                                    .Cells(3 + kCold, Col_Qen2 + 4) = coldPieceTEnd
                                    .Cells(3 + kCold, Col_Qen2 + 5) = qSeg
                                    kCold = kCold + 1
                                End If
                            End If
                        Next b
                    End If
                End If
            Next ii
        End If
ContinueHotSegment:
    Next i

    .Range("AL4:AS200").Sort Key1:=.Range("AM4"), Order1:=xlAscending, Header:=xlNo
    .Range("AU4:BB200").Sort Key1:=.Range("AV4"), Order1:=xlAscending, Header:=xlNo


End With

End Sub
